import requests

class RankLLMsClient:
    def __init__(self, base_url="https://rankllms-engine.onrender.com"):
        self.base_url = base_url

    def get_best_coding_model(self, min_context=128000):
        res = requests.get(f"{self.base_url}/api/v1/benchmarks?sort_by=coding_index&limit=20").json()
        for m in res.get("items", []):
            if m.get("context_length", 0) >= min_context:
                return m["slug"] # e.g. "anthropic/claude-opus-5"
        return "anthropic/claude-3.7-sonnet"

    def compare(self, model_a, model_b):
        return requests.get(f"{self.base_url}/api/v1/compare?model_a={model_a}&model_b={model_b}").json()
